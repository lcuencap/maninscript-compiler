laFaena pavos acumula(pavos limite) {
    unTal pavos total = 0;
    unTal pavos i = 0;

    segunVea (i < limite) {
        unTal pavos j = 0;

        segunVea (j < 3) {
            fijateSi ((i + j) > 2 y limite > 0) {
                total = total + i;
            } ySiNo {
                total = total + j;
            }
            j = j + 1;
        }

        i = i + 1;
    }

    suelta total;
}
